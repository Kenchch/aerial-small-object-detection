# torch/torchvision need the cu124 index for a GPU build (see README "Setup"):
#   pip install torch torchvision --index-url https://download.pytorch.org/whl/cu124
# Pinned to what was actually tested; an unpinned "torch" here would let pip
# resolve a newer release with no guarantee it still supports CUDA 12.4.
torch==2.6.0
torchvision==0.21.0

# ultralytics is pinned for the same reason torch is, and matters more: it owns
# the training loop, the augmentation defaults this project deliberately does
# not override, the letterbox implementation the label-scale analysis depends
# on, and the .val() pass that produces every mAP figure in the README. An
# unpinned ultralytics means the committed metrics are reproducible only by
# whoever installs on the same day.
ultralytics==8.4.110
onnx==1.22.0
onnxslim==0.1.94
onnxruntime-gpu==1.20.2  # CUDA 12 line -- see README engineering notes

# Read from the conda env this project runs in, not chosen -- and that env
# reproduces the committed metrics exactly (mAP50 0.3748 / mAP50-95 0.2216 from
# src/evaluate.py), which is the property worth claiming. Whether it is
# byte-for-byte the env of the original training run is not something the repo
# can prove, so it is not claimed.
#
# Note numpy is 2.x. The tempting pin is 1.26, on the theory that 2.x breaks
# ultralytics and opencv wheels; this project demonstrably runs on 2.4.4
# alongside the torch 2.6.0 pin above, so pinning back would pin a
# configuration none of the published figures were measured under.
opencv-python==5.0.0.93
numpy==2.4.4
Pillow==12.2.0
PyYAML==6.0.3
